  function calcSubnet() {

    const ip = document.getElementById('ipInput').value.trim();

    const mask = document.getElementById('maskInput').value.trim();

    let cidr;

    if (mask.startsWith('/')) cidr = parseInt(mask.slice(1));

    else {

      const parts = mask.split('.').map(Number);

      const bin = parts.map(p=>p.toString(2).padStart(8,'0')).join('');

      cidr = bin.split('').filter(b=>b==='1').length;

    }

    if (isNaN(cidr)||cidr<0||cidr>32) { alert('Invalid mask. Use /24 or 255.255.255.0'); return; }

    const parts = ip.split('.').map(Number);

    if (parts.length!==4||parts.some(p=>isNaN(p)||p<0||p>255)) { alert('Invalid IP address'); return; }


    const totalBits = 32;

    const hostBits = totalBits - cidr;

    const totalHosts = Math.pow(2,hostBits);

    const usable = Math.max(totalHosts-2,0);


    const maskNum = cidr===0 ? 0 : (0xffffffff << (32-cidr)) >>> 0;

    const ipNum = ((parts[0]<<24)|(parts[1]<<16)|(parts[2]<<8)|parts[3]) >>> 0;

    const netNum = (ipNum & maskNum) >>> 0;

    const bcastNum = (netNum | (~maskNum >>> 0)) >>> 0;

    const firstNum = cidr<31 ? netNum+1 : netNum;

    const lastNum = cidr<31 ? bcastNum-1 : bcastNum;

    const wildNum = (~maskNum) >>> 0;


    const toIP = n => [(n>>>24)&255,(n>>>16)&255,(n>>>8)&255,n&255].join('.');

    const toBin = n => [(n>>>24)&255,(n>>>16)&255,(n>>>8)&255,n&255].map(b=>b.toString(2).padStart(8,'0')).join('.');


    const results = [

      { label:'Network Address', value: toIP(netNum) },

      { label:'Subnet Mask', value: toIP(maskNum) },

      { label:'Wildcard Mask', value: toIP(wildNum) },

      { label:'Broadcast', value: toIP(bcastNum) },

      { label:'First Host', value: toIP(firstNum) },

      { label:'Last Host', value: toIP(lastNum) },

      { label:'Total Hosts', value: totalHosts.toLocaleString() },

      { label:'Usable Hosts', value: usable.toLocaleString() },

    ];


    const el = document.getElementById('subnetResult');

    el.innerHTML = results.map(r=>`

      <div style="background:var(--surface2);border:1px solid var(--border);border-radius:10px;padding:12px 14px">

        <div style="font-family:'Syne',sans-serif;font-size:14px;font-weight:700;color:var(--accent1)">${r.value}</div>

      </div>

    `).join('');

  }
